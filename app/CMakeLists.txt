cmake_minimum_required(VERSION 3.24...4.4)
project(TolkeyenAudioguia VERSION 0.1.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
    message(FATAL_ERROR "Use the portable Windows toolchain. See scripts/build.ps1")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "This release must be built with an x86 compiler")
endif()
add_compile_definitions(WINVER=0x0601 _WIN32_WINNT=0x0601 NOMINMAX UNICODE _UNICODE)
set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE STRING "" FORCE)
set(wxBUILD_TESTS OFF CACHE STRING "" FORCE)
set(wxBUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(wxBUILD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(wxBUILD_INSTALL OFF CACHE BOOL "" FORCE)
set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
set(wxUSE_WINRT OFF CACHE BOOL "" FORCE)
set(wxUSE_AUI OFF CACHE BOOL "" FORCE)
set(wxUSE_RIBBON OFF CACHE BOOL "" FORCE)
set(wxUSE_RICHTEXT OFF CACHE BOOL "" FORCE)
set(wxUSE_STC OFF CACHE BOOL "" FORCE)
set(wxUSE_PROPGRID OFF CACHE BOOL "" FORCE)
set(wxUSE_XRC OFF CACHE BOOL "" FORCE)
set(wxUSE_HTML OFF CACHE BOOL "" FORCE)
set(wxUSE_WXHTML_HELP OFF CACHE BOOL "" FORCE)
set(wxUSE_OPENGL OFF CACHE BOOL "" FORCE)
set(wxUSE_WEBREQUEST OFF CACHE BOOL "" FORCE)
set(wxUSE_SYS_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor build-wx EXCLUDE_FROM_ALL)
add_library(audioguia_core STATIC
    src/domain/Models.cpp src/domain/PlaybackPlanner.cpp
    src/config/ConfigRepository.cpp src/app/AppController.cpp)
target_include_directories(audioguia_core PUBLIC src PRIVATE vendor)
target_compile_options(audioguia_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(audioguia_audio STATIC src/audio/MiniaudioPlayer.cpp)
target_include_directories(audioguia_audio PUBLIC src)
target_link_libraries(audioguia_audio PUBLIC audioguia_core ole32)
add_executable(TolkeyenAudioguia WIN32
    src/app/App.cpp src/app/App.rc src/input/KeyboardController.cpp
    src/ui/MainFrame.cpp
    src/ui/SequenceEditor.cpp src/ui/PreferencesDialog.cpp)
target_compile_options(TolkeyenAudioguia PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(TolkeyenAudioguia PRIVATE src/app)
target_link_libraries(TolkeyenAudioguia PRIVATE audioguia_core audioguia_audio wx::core wx::base)
set_target_properties(TolkeyenAudioguia PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_options(TolkeyenAudioguia PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--major-subsystem-version,6,--minor-subsystem-version,1)
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE audioguia_core)
    target_link_options(core_tests PRIVATE -static)
    add_test(NAME core COMMAND core_tests)
    add_executable(controller_tests tests/controller_tests.cpp)
    target_link_libraries(controller_tests PRIVATE audioguia_core)
    target_link_options(controller_tests PRIVATE -static)
    add_test(NAME controller COMMAND controller_tests)
    add_executable(keyboard_tests tests/keyboard_tests.cpp)
    target_link_libraries(keyboard_tests PRIVATE audioguia_core)
    target_link_options(keyboard_tests PRIVATE -static)
    add_test(NAME keyboard COMMAND keyboard_tests)
    add_executable(audio_tests tests/audio_tests.cpp)
    target_link_libraries(audio_tests PRIVATE audioguia_audio)
    target_link_options(audio_tests PRIVATE -static)
    add_test(NAME audio COMMAND audio_tests)
    set_tests_properties(audio PROPERTIES TIMEOUT 90)
endif()
